price_labels <- c("Barata", "Media", "Cara")

datos$SalePrice_Category <- cut(datos$SalePrice,

                                labels = price_labels,
                                include.lowest = TRUE)

table(datos$SalePrice_Category)

## Barata  Media   Cara

index <- createDataPartition(datos$SalePrice_Category, p = 0.8, list = FALSE)
train_data <- datos[index, ]
test_data <- datos[-index, ]
train_data$SalePrice_Category <- as.factor(train_data$SalePrice_Category)
test_data$SalePrice_Category <- as.factor(test_data$SalePrice_Category)

table(train_data$SalePrice_Category)

## Barata  Media   Cara

numeric_cols <- sapply(train_data, is.numeric)
numeric_cols_names <- names(train_data)[numeric_cols]

numeric_cols_names <- setdiff(numeric_cols_names, "SalePrice")

normalize <- function(x) {
  return((x - min(x, na.rm = TRUE)) / (max(x, na.rm = TRUE) - min(x, na.rm = TRUE)))
}

train_data_norm <- train_data
test_data_norm <- test_data

for (col in numeric_cols_names) {
  col_min <- min(train_data[[col]], na.rm = TRUE)
  col_max <- max(train_data[[col]], na.rm = TRUE)
  
  train_data_norm[[col]] <- (train_data[[col]] - col_min) / (col_max - col_min)
  test_data_norm[[col]] <- (test_data[[col]] - col_min) / (col_max - col_min)
}

for (col in numeric_cols_names) {
  mean_val <- mean(train_data_norm[[col]], na.rm = TRUE)
  train_data_norm[[col]][is.na(train_data_norm[[col]])] <- mean_val
  test_data_norm[[col]][is.na(test_data_norm[[col]])] <- mean_val
}

train_dummies <- model.matrix(~ SalePrice_Category - 1, data = train_data_norm)
train_data_norm <- cbind(train_data_norm, train_dummies)

test_dummies <- model.matrix(~ SalePrice_Category - 1, data = test_data_norm)
test_data_norm <- cbind(test_data_norm, test_dummies)
predictors <- setdiff(numeric_cols_names, c("SalePrice", "SalePrice_Category"))

formula_nn <- as.formula(paste("SalePrice_Category ~ ", paste(predictors, collapse = " + ")))

# MODELO 1 
start_time_model1 <- Sys.time()
set.seed(2075)
nn_model1 <- nnet(
  formula_nn,
  data = train_data_norm,
  size = 5,            
  decay = 0.01,        
  maxit = 1000,        
  linout = FALSE,      
  trace = FALSE        
)
end_time_model1 <- Sys.time()

Explicacion de los modelos y sus resultados

Para el análisis de los datos de precios de viviendas se hizo la comparación entre dos modelos de redes neuronales, el Modelo 1 con 5 neuronas y decay de 0.01, y el Modelo 2 con 10 neuronas y decay de 0.001. Ademas que el modelo 2, no tiene conexiones directas. Los resultados muestran que el Modelo 1 alcanzó una exactitud del 80.4%, superando al Modelo 2 que logró un 77.7%. Además, el Modelo 1 fue significativamente más rápido en el entrenamiento, tomando solo 0.27 segundos frente a 1.64 segundos del Modelo 2 (Tomando en cuenta que los resultados suelen variar cada vez que se corre el Rmd, pero son bastante similares siempre). Esta comparación inicial sugiere que la arquitectura más simple, modelo 1, es más efectiva para este problema de clasificación.

La matriz de confusión del modelo optimizado muestra mejoras notables. El modelo ajustado alcanza una sensibilidad del 81.3% para casas baratas, 75.0% para casas de precio medio y un 90.9% para casas caras. Particularmente, el modelo ya no confunde casas baratas con caras, y esto nos dice que ha aprendido a distinguir mejor entre los extremos de precios, aunque sigue teniendo cierta dificultad con la categoría intermedia.

La comparación entre el modelo original y el ajustado demuestra una mejora significativa. El modelo ajustado alcanza una exactitud del 82.5% frente al 80.4% del Modelo 1 origina. Más importante aún, el sobreajuste se reduce drásticamente del 12.6% en el modelo original a solo 3.3% en el modelo ajustado.

La gráfica final compara los tres modelos en términos de exactitud, kappa y sobreajuste. Y con la gráfica podemos confirmar que el Modelo Ajustado supera a los otros en exactitud y kappa, mientras mantiene un sobreajuste mínimo.
