train <- read.csv("train.csv")

set.seed(245)
porciento <- 0.7

lower_limit <- median(train_data$SalePrice)

                        ifelse(train_data$SalePrice > (lower_limit + upper_limit), "Caras", "Medianas"))

test_data$Category <- ifelse(test_data$SalePrice < (lower_limit - upper_limit), "Baratas",
                       ifelse(test_data$SalePrice > (lower_limit + upper_limit), "Caras", "Medianas"))

numeric_vars <- c("OverallQual", "GrLivArea", "GarageCars", "TotalBsmtSF", 
                 "FullBath", "YearBuilt", "YearRemodAdd", "Fireplaces",
                 "BsmtFinSF1", "LotArea", "MasVnrArea", "WoodDeckSF")

for(var in numeric_vars) {
  if(sum(is.na(train_data[[var]])) > 0) {
    median_value <- median(train_data[[var]], na.rm = TRUE)
    train_data[[var]][is.na(train_data[[var]])] <- median_value
  }
  
  if(sum(is.na(test_data[[var]])) > 0) {
    median_value <- median(train_data[[var]], na.rm = TRUE)
    test_data[[var]][is.na(test_data[[var]])] <- median_value
  }
}

for(var in categorical_vars) {
  if(sum(is.na(train_data[[var]])) > 0) {
    mode_value <- names(sort(table(train_data[[var]]), decreasing = TRUE))[1]
    train_data[[var]][is.na(train_data[[var]])] <- mode_value
  }
  
  if(sum(is.na(test_data[[var]])) > 0) {
    mode_value <- names(sort(table(train_data[[var]]), decreasing = TRUE))[1]
    test_data[[var]][is.na(test_data[[var]])] <- mode_value
  }
  
  all_levels <- unique(c(levels(factor(train_data[[var]])), levels(factor(test_data[[var]]))))
  train_data[[var]] <- factor(train_data[[var]], levels = all_levels)
  test_data[[var]] <- factor(test_data[[var]], levels = all_levels)
}

evaluate_model_sizes <- function(X, y, sizes, seed = 245) {
  train_errors <- numeric(length(sizes))
  test_errors <- numeric(length(sizes))
  
  for (i in seq_along(sizes)) {
    set.seed(seed)
    size <- sizes[i]
    sample_indices <- sample(nrow(X), size = size)
    
    X_train_subset <- X[sample_indices, ]
    y_train_subset <- y[sample_indices]
    
    X_val <- X[-sample_indices, ]
    y_val <- y[-sample_indices]
    
    model <- glm(y_train_subset ~ ., data = as.data.frame(X_train_subset), family = "binomial")
    
    train_probs <- predict(model, newdata = as.data.frame(X_train_subset), type = "response")
    train_preds <- ifelse(train_probs > 0.5, 1, 0)
    train_errors[i] <- mean(train_preds != y_train_subset)
    
    test_probs <- predict(model, newdata = as.data.frame(X_val), type = "response")
    test_preds <- ifelse(test_probs > 0.5, 1, 0)
    test_errors[i] <- mean(test_preds != y_val)
  }
  
  return(list(train_errors = train_errors, test_errors = test_errors))
}
